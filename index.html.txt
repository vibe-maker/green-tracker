    <script>
        // 가상의 학생 데이터 (초기 로컬 저장소 세팅용)
        let currentUser = "";
        let studentData = JSON.parse(localStorage.getItem('greenTrackerData')) || {};

        function login() {
            const no = document.getElementById('user-no').value;
            const pw = document.getElementById('user-pw').value;

            if(!no || !pw) { alert("번호와 비밀번호를 모두 입력하세요."); return; }
            
            // 번호를 학생 ID로 활용 (예: student_5)
            currentUser = "student_" + no;
            
            if(!studentData[currentUser]) {
                studentData[currentUser] = {
                    password: pw,
                    scores: {} // 날짜별 점수 저장용
                };
            } else if (studentData[currentUser].password !== pw) {
                alert("비밀번호가 일치하지 않습니다!");
                return;
            }

            localStorage.setItem('greenTrackerData', JSON.stringify(studentData));
            document.getElementById('login-box').style.display = 'none';
            document.getElementById('main-content').style.display = 'block';
            document.getElementById('welcome-msg').innerText = `👋 반갑습니다, ${no}번 지킴이 어린이!`;
            
            initCalendar();
            loadTodayData();
        }

        // 급식판 슬롯 토글
        function toggleFood(element) {
            element.classList.toggle('leftover');
            calculateScore();
        }

        // 점수 계산 핵심 로직
        function calculateScore() {
            // 1. 급식 잔반 점수 계산
            const leftovers = document.querySelectorAll('.food-slot.leftover').length;
            const leftoverPercent = leftovers * 20;
            document.getElementById('meal-percent').innerText = `음식 남김 정도: ${leftoverPercent}% (감점 ${leftoverPercent}점)`;

            // 2. 기타 생활 습관 점수 계산
            const hasCup = document.getElementById('cup-check').checked ? 10 : 0;
            const hasMultiCup = document.getElementById('multi-cup-check').checked ? 10 : 0;
            const hasPlant = document.getElementById('plant-check').checked ? 10 : 0;
            const hasRecycle = document.getElementById('recycle-check').checked ? 100 : 0;

            // 전체 점수 = 분리수거 점수(100/0) + 양치컵(10) + 물컵(10) + 식물(10) - 잔반 감점(최대 100)
            let totalScore = hasRecycle + hasCup + hasMultiCup + hasPlant - leftoverPercent;
            
            // 점수가 음수로 떨어지지 않도록 보정
            if (totalScore < 0) totalScore = 0;

            document.getElementById('total-score').innerText = totalScore + "점";
            return totalScore;
        }

        // 데이터 저장
        function saveToday() {
            const score = calculateScore();
            const today = new Date().getDate(); // 오늘 날짜 (일자만 간단히 활용)

            studentData[currentUser].scores[today] = score;
            localStorage.setItem('greenTrackerData', JSON.stringify(studentData));
            
            alert("오늘의 초록 점수가 성공적으로 저장되었습니다!");
            initCalendar(); // 달력 업데이트
        }

        // 저장된 데이터 불러오기
        function loadTodayData() {
            const today = new Date().getDate();
            const userScores = studentData[currentUser].scores;

            if(userScores && userScores[today] !== undefined) {
                // 오늘 저장한 기록이 있다면 복원 시뮬레이션 (간소화 버젼)
                document.getElementById('total-score').innerText = userScores[today] + "점";
            }
            calculateScore();
        }

        // 달력 그리기 및 누적 점수 계산
        function initCalendar() {
            const calendarBoard = document.getElementById('calendar-board');
            calendarBoard.innerHTML = ""; // 초기화

            // 요일 헤더 생성
            const weeks = ['일', '월', '화', '수', '목', '금', '토'];
            weeks.forEach(w => {
                const header = document.createElement('div');
                header.className = 'calendar-header';
                header.innerText = w;
                calendarBoard.appendChild(header);
            });

            const totalDays = 31; // 5월 기준 31일
            const userScores = studentData[currentUser].scores || {};
            let cumulativeScore = 0;

            for (let i = 1; i <= totalDays; i++) {
                const dayDiv = document.createElement('div');
                dayDiv.className = 'calendar-day';
                
                const numSpan = document.createElement('span');
                numSpan.className = 'day-num';
                numSpan.innerText = i;
                dayDiv.appendChild(numSpan);

                // 기록된 점수가 있다면 표시
                if (userScores[i] !== undefined) {
                    const scoreSpan = document.createElement('span');
                    scoreSpan.className = 'day-score';
                    scoreSpan.innerText = userScores[i] + "점";
                    dayDiv.appendChild(scoreSpan);
                    cumulativeScore += userScores[i];
                }

                calendarBoard.appendChild(dayDiv);
            }

            document.getElementById('cumulative-score').innerText = `누적 총 점수: ${cumulativeScore}점`;
        }
    </script>
